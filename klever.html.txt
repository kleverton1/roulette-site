        // Track daily stats for winner
        function trackWinnerStats(winnerUsername, totalTickets, prizePool) {
            const winPercentage = ((gameState.playerDetails[winnerUsername].totalBet / totalTickets) * 100).toFixed(2);
            const winAmount = prizePool;
            
            // Add to daily winners list
            gameState.dailyStats.winners.push({
                username: winnerUsername,
                avatar: gameState.playerDetails[winnerUsername].avatar,
                frame: gameState.playerDetails[winnerUsername].frame,
                winPercentage: winPercentage,
                winAmount: winAmount,
                timestamp: new Date()
            });
            
            // Update Везунчик дня (player with smallest win percentage)
            if (!gameState.dailyStats.minWinPercentage || 
                parseFloat(winPercentage) < parseFloat(gameState.dailyStats.minWinPercentage)) {
                gameState.dailyStats.minWinPercentage = winPercentage;
                gameState.dailyStats.minWinPlayer = {
                    username: winnerUsername,
                    avatar: gameState.playerDetails[winnerUsername].avatar,
                    frame: gameState.playerDetails[winnerUsername].frame,
                    winPercentage: winPercentage,
                    winAmount: winAmount
                };
            }
            
            // Update JackPot дня (player with largest win amount)
            if (winAmount > gameState.dailyStats.maxWinAmount) {
                gameState.dailyStats.maxWinAmount = winAmount;
                gameState.dailyStats.maxWinPlayer = {
                    username: winnerUsername,
                    avatar: gameState.playerDetails[winnerUsername].avatar,
                    frame: gameState.playerDetails[winnerUsername].frame,
                    winPercentage: winPercentage,
                    winAmount: winAmount
                };
            }
            
            // Update Последний победитель
            gameState.dailyStats.lastWinner = {
                username: winnerUsername,
                avatar: gameState.playerDetails[winnerUsername].avatar,
                frame: gameState.playerDetails[winnerUsername].frame,
                winPercentage: winPercentage,
                winAmount: winAmount
            };
            
            // Update the display
            updateDailyWinners();
        }
        
        // Get random frame type
        function getRandomFrame() {
            if (Math.random() < 0.6) {

            return gameState.frameTypes[Math.floor(Math.random() * (gameState.frameTypes.length - 1)) + 1];
        }
        
        // Add a demo player
        function addDemoPlayer(username, amount, avatarUrl, frameType) {
            if (!gameState.playerDetails[username]) {
                gameState.playerDetails[username] = {
                    totalBet: 0,
                    avatar: avatarUrl,
                    frame: frameType || 'frame-default'
                };
            }
            
            gameState.playerDetails[username].totalBet += amount;
            gameState.prizePool += amount;
            
            // Add to players array for each TON
            for (let i = 0; i < amount; i++) {
                gameState.players.push({
                    username: username,
                    avatar: avatarUrl,
                    frame: frameType || 'frame-default'
                });
            }
            
            updatePlayersList();
            updateUI();
            
            // If we have at least 2 players, start countdown
            if (Object.keys(gameState.playerDetails).length >= 2 && !gameState.roundActive) {
                startCountdown();
            }
        }
        
        // Update players list
        function updatePlayersList() {
            playersList.innerHTML = '';
            let totalTickets = 0;
            
            for (const username in gameState.playerDetails) {
                totalTickets += gameState.playerDetails[username].totalBet;
            }
            
            // Sort players by bet amount (descending)
            const sortedPlayers = Object.keys(gameState.playerDetails)
                .map(username => ({
                    username,
                    ...gameState.playerDetails[username]
                }))
                .sort((a, b) => b.totalBet - a.totalBet);
            
            for (const player of sortedPlayers) {
                const winChance = ((player.totalBet / totalTickets) * 100).toFixed(1);
                
                const playerItem = document.createElement('div');
                playerItem.className = 'player-item';
                playerItem.innerHTML = `
                    <div class="player-avatar ${player.frame}">
                        <img src="${player.avatar}" class="w-full h-full rounded-lg">
                    </div>
                    <div class="player-info">
                        <div class="player-username">${player.username}</div>
                        <div class="player-bet">${player.totalBet} TON</div>
                    </div>
                    <div class="player-chance">${winChance}%</div>
                `;
                playersList.appendChild(playerItem);
            }
            
            playersCount.textContent = Object.keys(gameState.playerDetails).length;
        }
        
        // Start the countdown to round start
        function startCountdown() {
            gameState.roundActive = true;
            countdownElement.classList.remove('hidden');
            roundStatusElement.textContent = 'Раунд начинается...';
            
            const countdownInterval = setInterval(() => {
                gameState.countdown--;
                countdownElement.textContent = gameState.countdown;
                
                if (gameState.countdown <= 0) {
                    clearInterval(countdownInterval);
                    startRound();
                }
            }, 1000);
        }

                    roundStatusElement.classList.add('text-yellow-400');

        // Reset round for next game
        function resetRound() {
            gameState.roundActive = false;
            gameState.players = [];
            gameState.playerDetails = {};
            gameState.countdown = 15;
            gameState.prizePool = 0;
            rouletteTrack.innerHTML = '';
            rouletteTrack.style.transform = 'translateX(0)';
            roundStatusElement.textContent = 'Ожидание игроков...';
            roundStatusElement.classList.remove('text-yellow-400');
            playersList.innerHTML = '';
            playersCount.textContent = '0';
            
            // Enable bet button for new round
            placeBetBtn.classList.remove('btn-disabled');
            
            updateUI();
        }
        
        // Place a bet
        function placeBet() {
            const betAmount = parseInt(betAmountElement.value);
            
            if (betAmount < 1 || betAmount > gameState.userBalance) {
                return;
            }
            
            gameState.userBalance -= betAmount;
            updateUI();
            
            const randomAvatar = demoAvatars[Math.floor(Math.random() * demoAvatars.length)];
            const username = '@you';
            if (!gameState.playerDetails[username]) {
                gameState.playerDetails[username] = {
                    totalBet: 0,
                    avatar: randomAvatar,
                    frame: gameState.userFrame || 'frame-default'
                };
            }
            
            gameState.playerDetails[username].totalBet += betAmount;
            gameState.prizePool += betAmount;
            
            for (let i = 0; i < betAmount; i++) {
                gameState.players.push({
                    username: username,
                    avatar: randomAvatar,
                    frame: gameState.userFrame || 'frame-default'
                });
            }
            
            updatePlayersList();
            updateUI();
            
            // If we have at least 2 players, start countdown
            if (Object.keys(gameState.playerDetails).length >= 2 && !gameState.roundActive) {
                startCountdown();
            }
        }
        
        // Buy item from shop
        function buyItem(itemType, price) {
            if (gameState.userBalance < price) {
                alert('Недостаточно средств!');
                return;
            }
            
            if (itemType.startsWith('frame-')) {
                gameState.userFrame = itemType;
                alert(`Вы приобрели рамку: ${itemType.replace('frame-', '').toUpperCase()}`);
            }
            
            gameState.userBalance -= price;
            updateUI();
        }
        
        // Adjust bet amount
        function adjustBet(change) {
            let newBet = parseInt(betAmountElement.value) + change;
            newBet = Math.max(1, Math.min(newBet, gameState.userBalance));
            betAmountElement.value = newBet;
        }
        
        // Update UI elements
        function updateUI() {
            poolAmountElement.textContent = gameState.prizePool;
            userBalanceElement.textContent = gameState.userBalance;
            userBalanceElement2.textContent = gameState.userBalance;
            betAmountElement.max = gameState.userBalance;
        }
